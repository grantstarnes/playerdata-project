library(tidyverse)

samp <- read_csv("https://docs.google.com/spreadsheets/d/e/2PACX-1vTJ4oZBg00Y9kg71OvLI6Md3hzmJuSf1GMUaih8tt2Opryf_NB30T8nHas_GY4RQ_q33jLXjNDcDNgE/pub?gid=1727941844&single=true&output=csv")

w_soc <- samp |>
  filter(athlete_sport == "association_football",
         athlete_gender_marker == "Female",
         active_minutes >= 70)

# Non-Weighted Averages

avgs <- w_soc |>
  group_by(athlete_relative_age) |>
  summarize(count = n(),
            avg_total_distance_m = mean(total_distance_m),
            avg_max_speed_kph = mean(max_speed_kph),
            avg_session_load = mean(session_load),
            avg_sprint_events = mean(sprint_events),
            sd_total_distance_m = sd(total_distance_m),
            sd_max_speed_kph = sd(max_speed_kph),
            sd_session_load = sd(session_load),
            sd_sprint_events = sd(sprint_events),
            #top10_t_dist = quantile(total_distance_m, probs = 0.9, na.rm = T),
            #bottom10_t_dist = quantile(total_distance_m, probs = 0.1, na.rm = T),
            #top10_max_speed_kph = quantile(max_speed_kph, probs = 0.9, na.rm = T),
            #bottom10_max_speed_kph = quantile(max_speed_kph, probs = 0.1, na.rm = T),
            #top10_session_load = quantile(session_load, probs = 0.9, na.rm = T),
            #bottom10_session_load = quantile(session_load, probs = 0.1, na.rm = T),
            top10_sprint_events = quantile(sprint_events, probs = 0.9, na.rm = T),
            bottom10_sprint_events = quantile(sprint_events, probs = 0.1, na.rm = T)
            )

w_soc <- w_soc |>
  group_by(athlete_relative_age) |>
  mutate(total_dist_perc_rank = percent_rank(total_distance_m) * 100,
  max_speed_perc_rank = percent_rank(max_speed_kph) * 100,
  session_load_perc_rank = percent_rank(session_load) * 100) |>
  ungroup()

w_soc_select <- w_soc |>
  select(athlete_number,
  athlete_relative_age,
  active_minutes,
  total_distance_m,
  max_speed_kph,
  session_load,
  total_dist_perc_rank,
  max_speed_perc_rank,
  session_load_perc_rank)

# Weighted Averages

weighted_avgs <- w_soc |>
  group_by(athlete_relative_age) |>
  summarize(count = n(),
            weighted_avg_total_distance_m = weighted.mean(total_distance_m, active_minutes),
            weighted_avg_max_speed_kph = weighted.mean(max_speed_kph, active_minutes),
            weighted_avg_session_load = weighted.mean(session_load, active_minutes),
            sd_total_distance_m = sd(total_distance_m),
            sd_max_speed_kph = sd(max_speed_kph),
            sd_session_load = sd(session_load),
            top10_t_dist = quantile(total_distance_m, probs = 0.9, na.rm = T),
            bottom10_t_dist = quantile(total_distance_m, probs = 0.1, na.rm = T),
            top10_max_speed_kph = quantile(max_speed_kph, probs = 0.9, na.rm = T),
            bottom10_max_speed_kph = quantile(max_speed_kph, probs = 0.1, na.rm = T),
            top10_session_load = quantile(session_load, probs = 0.9, na.rm = T),
            bottom10_session_load = quantile(session_load, probs = 0.1, na.rm = T),
            )

# Creating session_id

w_soc_select_2 <- w_soc_select |>
  group_by(athlete_number) |>
  mutate(session_id = row_number())

# Aggregating into Singular Row per Athlete

weighted_avg_ath <- w_soc_select_2 |>
  group_by(athlete_number, athlete_relative_age) |>
  summarise(
    w_total_distance = weighted.mean(total_distance_m, active_minutes),
    w_session_load   = weighted.mean(session_load, active_minutes),
    w_max_speed_kph = weighted.mean(max_speed_kph, active_minutes)
  )

weighted_avg_ath <- weighted_avg_ath |>
  group_by(athlete_relative_age) |>
  mutate(total_dist_perc_rank = percent_rank(w_total_distance) * 100,
  max_speed_perc_rank = percent_rank(w_max_speed_kph) * 100,
  session_load_perc_rank = percent_rank(w_session_load) * 100,
  num_sessions = n()) |>
  ungroup()

  weighted_avg_ath
